rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && request.auth.token.email == "jbonilla2987@gmail.com";
    }

    function isApproved(userId) {
      return isAdmin()
        || (
          signedIn()
          && request.auth.uid == userId
          && exists(/databases/$(database)/documents/accessRequests/$(userId))
          && get(/databases/$(database)/documents/accessRequests/$(userId)).data.status == "approved"
        );
    }

    match /accessRequests/{userId} {
      allow list: if isAdmin();
      allow get: if isAdmin() || (signedIn() && request.auth.uid == userId);

      allow create: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && request.resource.data.status == "pending"
        && request.resource.data.role == "user"
        && request.resource.data.email == request.auth.token.email
      );

      allow update, delete: if isAdmin();
    }

    // Solicitud de eliminación: el usuario solo solicita; el administrador aprueba o rechaza.
    match /dataDeletionRequests/{userId} {
      allow list: if isAdmin();
      allow get: if isAdmin() || (signedIn() && request.auth.uid == userId);
      allow create, update: if (
        signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.status == "pending"
      ) || isAdmin();
      allow delete: if isAdmin();
    }

    // Ubicaciones GPS compartidas desde el celular.
    match /liveLocations/{userId} {
      allow read: if isAdmin();
      allow create, update: if signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.lat is number
        && request.resource.data.lng is number;
      allow delete: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
      );
    }

    // Directorio mínimo para el chat entre usuarios. No expone correo electrónico.
    match /chatDirectory/{userId} {
      allow read: if isAdmin() || (signedIn() && isApproved(request.auth.uid));
      allow create, update: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
        && request.resource.data.uid == request.auth.uid
      );
      allow delete: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
    }

    // Soporte privado: cada usuario solo ve su conversación; el administrador puede ver todas.
    match /supportThreads/{userId} {
      allow list: if isAdmin();
      allow get: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
      allow create, update: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
      allow delete: if isAdmin();

      match /messages/{messageId} {
        allow read: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
        allow create: if (
          isAdmin()
          || (signedIn() && request.auth.uid == userId && isApproved(userId))
        ) && request.resource.data.senderUid == request.auth.uid;
        allow update, delete: if isAdmin();
      }
    }

    // Conversaciones privadas entre dos usuarios aprobados.
    match /userChats/{chatId} {
      allow read: if isAdmin() || (
        signedIn()
        && isApproved(request.auth.uid)
        && resource.data.participants is list
        && resource.data.participants.hasAny([request.auth.uid])
      );

      allow create: if signedIn()
        && isApproved(request.auth.uid)
        && request.resource.data.participants is list
        && request.resource.data.participants.size() == 2
        && request.resource.data.participants.hasAny([request.auth.uid]);

      allow update: if isAdmin() || (
        signedIn()
        && isApproved(request.auth.uid)
        && resource.data.participants.hasAny([request.auth.uid])
        && request.resource.data.participants == resource.data.participants
      );
      allow delete: if isAdmin();

      match /messages/{messageId} {
        allow read: if isAdmin() || (
          signedIn()
          && isApproved(request.auth.uid)
          && get(/databases/$(database)/documents/userChats/$(chatId)).data.participants.hasAny([request.auth.uid])
        );
        allow create: if signedIn()
          && isApproved(request.auth.uid)
          && get(/databases/$(database)/documents/userChats/$(chatId)).data.participants.hasAny([request.auth.uid])
          && request.resource.data.senderUid == request.auth.uid;
        allow update, delete: if isAdmin();
      }
    }

    // Auditoría de acciones sensibles. Solo el administrador puede consultarla.
    // Las Cloud Functions usan Admin SDK y no dependen de estas reglas para escribir.
    match /adminAuditLogs/{logId} {
      allow read: if isAdmin();
      allow create, update, delete: if false;
    }

    // Datos financieros privados de cada usuario.
    match /users/{userId} {
      allow read: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
      allow create, update: if signedIn()
        && request.auth.uid == userId
        && isApproved(userId);
      allow delete: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
      );

      match /{document=**} {
        allow read: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
        allow create, update: if signedIn()
          && request.auth.uid == userId
          && isApproved(userId);
        allow delete: if isAdmin() || (
          signedIn()
          && request.auth.uid == userId
          && isApproved(userId)
        );
      }
    }
  }
}
