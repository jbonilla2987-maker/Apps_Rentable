rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && request.auth.token.email == "jbonilla2987@gmail.com";
    }

    function isApproved(userId) {
      return isAdmin()
        || (
          signedIn()
          && request.auth.uid == userId
          && exists(/databases/$(database)/documents/accessRequests/$(userId))
          && get(/databases/$(database)/documents/accessRequests/$(userId)).data.status == "approved"
        );
    }

    match /accessRequests/{userId} {
      allow list: if isAdmin();
      allow get: if isAdmin() || (signedIn() && request.auth.uid == userId);

      allow create: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && request.resource.data.status == "pending"
        && request.resource.data.role == "user"
        && request.resource.data.email == request.auth.token.email
      );

      allow update: if isAdmin();
      // Un usuario aprobado puede retirar su propia solicitud al eliminar su cuenta.
      allow delete: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && resource.data.status == "approved"
      );
    }

    // Ubicaciones GPS compartidas voluntariamente desde el celular.
    match /liveLocations/{userId} {
      // Solo el administrador puede ver el mapa general.
      allow read: if isAdmin();

      // Cada usuario aprobado solo puede publicar/actualizar su propia ubicación.
      allow create, update: if signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.lat is number
        && request.resource.data.lng is number;

      allow delete: if isAdmin() || (
        signedIn()
        && request.auth.uid == userId
        && isApproved(userId)
      );
    }

    // Metadatos de la cuenta.
    match /users/{userId} {
      // El administrador puede leer los datos para generar respaldos JSON por usuario.
      // Solo el propietario aprobado puede crear, modificar o borrar sus datos.
      allow read: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
      allow create, update, delete: if signedIn()
        && request.auth.uid == userId
        && isApproved(userId);

      // Jornadas, gastos, vehículos y configuración quedan en documentos separados.
      match /{document=**} {
        allow read: if isAdmin() || (signedIn() && request.auth.uid == userId && isApproved(userId));
        allow create, update, delete: if signedIn()
          && request.auth.uid == userId
          && isApproved(userId);
      }
    }
  }
}
